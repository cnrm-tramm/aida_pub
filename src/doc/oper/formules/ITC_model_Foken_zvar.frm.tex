\hline 
$ITC_W$ & $ \vert Z/L \vert < 0,032 $ & 1,3 &  0 \\ 
  & $\vert Z/L \vert \geq 0,032$ & 2,0 & 1/8 \\ 
\hline 
$ITC_U$ & $\vert Z/L \vert < 0,032$ & 2,7 &  0 \\ 
  & $\vert Z/L \vert \geq 0,032$ & 4,15 & 1/8 \\
\hline 
$ITC_T$ & $0,02 \leq Z/L  < 1$ & 1,4 & -1/4 \\ 
 & $0,02 > Z/L > -0,062 $ & 0,5 & -1/2 \\
 & $-0,062 > Z/L > -1 $ & 1,0 & -1/4 \\
 & $-1 > Z/L  $ & 1,0 & -1/3 \\
 & $Z/L \geq 1 $ & 1/4 & -1/3 \\
\hline 
\end{tabular}
\end{center}
