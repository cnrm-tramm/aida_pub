\section{Différenciation discrète centrée}

\itemstyle{Op\'erateur~:} \operfmt{Differ\_centre\_XfYd}
\index{Differ\_centre\_XfYd}
\label{OP-Differ_centre_XfYd}

%%%%%OBJET%%%%
\itemstyle{Objet~:}
Calcule de manière centrée le taux d'acroissement (ou dérivée partielle discrète) 
d'une variable de type \texttt{float} par rapport à une autre de type \texttt{double}.

%%%%%ARGUMENTS%%%%
\bigskip
\input{arguments/Differ_centre_XfYd.args.tex}

%%%%%FORMULE%%%%
\itemstyle{Formule}

\input{formules/Differ_centre_XfYd.frm.tex}

%%%%%AUTEUR%%%%
\itemstyle{Commentaire~:}
Renvoie une valeur manquante pour les vecteurs de taille 1.

Le calcul de la première et de la dernière valeur n'est pas centré.
